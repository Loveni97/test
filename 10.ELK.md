二、单机 Docker Compose 搭建
1. 创建目录
mkdir -p /opt/elk/{elasticsearch,logstash,kibana,filebeat}
cd /opt/elk


2. 修改系统参数

sysctl -w vm.max_map_count=262144

echo "vm.max_map_count=262144" >> /etc/sysctl.conf


3. docker-compose.yml

services:
  elasticsearch:
    image: docker.elastic.co/elasticsearch/elasticsearch:9.4.2
    container_name: elasticsearch
    environment:
      - discovery.type=single-node
      - ES_JAVA_OPTS=-Xms2g -Xmx2g
      - xpack.security.enabled=false
    ports:
      - "9200:9200"
    volumes:
      - esdata:/usr/share/elasticsearch/data

  logstash:
    image: docker.elastic.co/logstash/logstash:9.4.2
    container_name: logstash
    ports:
      - "5044:5044"
      - "9600:9600"
    volumes:
      - ./logstash/logstash.conf:/usr/share/logstash/pipeline/logstash.conf
    depends_on:
      - elasticsearch

  kibana:
    image: docker.elastic.co/kibana/kibana:9.4.2
    container_name: kibana
    ports:
      - "5601:5601"
    environment:
      - ELASTICSEARCH_HOSTS=http://elasticsearch:9200
    depends_on:
      - elasticsearch

volumes:
  esdata:



三、配置 Logstash

vim /opt/elk/logstash/logstash.conf

input {
  beats {
    port => 5044
  }
}

filter {
  if [log][file][path] =~ "nginx" {
    grok {
      match => {
        "message" => '%{IPORHOST:client_ip} - - \[%{HTTPDATE:time}\] "%{WORD:method} %{URIPATHPARAM:request} HTTP/%{NUMBER:http_version}" %{NUMBER:status} %{NUMBER:body_bytes} "-" "%{GREEDYDATA:user_agent}" "-"'
      }
    }
  }
}

output {
  elasticsearch {
    hosts => ["http://elasticsearch:9200"]
    index => "system-log-%{+YYYY.MM.dd}"
  }

  stdout {
    codec => rubydebug
  }
}


四、启动 ELK

docker compose up -d
docker ps

Elasticsearch：http://服务器IP:9200
Kibana：http://服务器IP:5601
Logstash Beats端口：5044


五、安装 Filebeat 采集日志




1. 创建仓库：Rocky Linux / CentOS 安装

cat > /etc/yum.repos.d/elastic.repo <<EOF
[elastic-9.x]
name=Elastic repository for 9.x packages
baseurl=https://artifacts.elastic.co/packages/9.x/yum
gpgcheck=1
enabled=1
gpgkey=https://artifacts.elastic.co/GPG-KEY-elasticsearch
EOF


dnf install filebeat -y



  vim /etc/filebeat/filebeat.yml


filebeat.inputs:
  - type: filestream
    id: system-log
    enabled: true
    paths:
      - /var/log/messages
      - /var/log/secure
      - /var/log/nginx/access.log

output.logstash:
  hosts: ["192.168.146.109:5044"]


systemctl enable --now filebeat.service

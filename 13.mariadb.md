

创建业务数据库
CREATE DATABASE collect_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;

创建业务用户

CREATE USER 'collect_user'@'%' IDENTIFIED BY 'linux0224';

GRANT SELECT, INSERT, UPDATE, DELETE 
ON collect_db.* 
TO 'collect_user'@'%';

FLUSH PRIVILEGES;

配置文件优化

/etc/my.cnf或者/etc/mysql/my.cnf

[mysqld]
server-id=1

character-set-server=utf8mb4
collation-server=utf8mb4_general_ci

max_connections=1000
connect_timeout=10
wait_timeout=600
interactive_timeout=600

innodb_buffer_pool_size=4G
innodb_log_file_size=512M
innodb_flush_log_at_trx_commit=1

slow_query_log=1
long_query_time=1
slow_query_log_file=/var/log/mysql-slow.log

log_bin=mysql-bin
binlog_format=ROW
binlog_expire_logs_seconds=604800

修改完重启
systemctl restart mysqld

创建表

USE collect_db;

CREATE TABLE collect_data (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    device_id VARCHAR(64) NOT NULL COMMENT '设备ID或服务器ID',
    metric_name VARCHAR(64) NOT NULL COMMENT '指标名称',
    metric_value DECIMAL(18,4) NOT NULL COMMENT '指标值',
    collect_time DATETIME(3) NOT NULL COMMENT '采集时间',
    write_time TIMESTAMP(3) DEFAULT CURRENT_TIMESTAMP(3) COMMENT '写入时间',
    PRIMARY KEY (id),
    KEY idx_device_time (device_id, collect_time),
    KEY idx_metric_time (metric_name, collect_time),
    KEY idx_collect_time (collect_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='采集数据表';


插入示例：
INSERT INTO collect_data 
(device_id, metric_name, metric_value, collect_time)
VALUES
('server-01', 'cpu_usage', 35.6, NOW(3));




## 主从复制配置

1. 修改主库配置文件
vim /etc/my.cnf.d/server.cnf

[mysqld]
server-id=1
log_bin=mariadb-bin
binlog_format=ROW
bind-address=0.0.0.0
expire_logs_days=7


systemctl restart mariadb

检查是否开启
mysql -uroot -p

SHOW VARIABLES LIKE 'log_bin';
SHOW MASTER STATUS;

+--------------------+----------+--------------+------------------+
| File               | Position | Binlog_Do_DB | Binlog_Ignore_DB |
+--------------------+----------+--------------+------------------+
| mariadb-bin.000001 |      342 |              |                  |
+--------------------+----------+--------------+------------------+

2. 主库创建复制用户

CREATE USER 'repl'@'192.168.1.11' IDENTIFIED BY 'Repl@123456';

GRANT REPLICATION SLAVE ON *.* TO 'repl'@'192.168.1.11';

FLUSH PRIVILEGES;

在从库上测试
mysql -h192.168.1.10 -urepl -p



3. 从库配置

vim /etc/my.cnf.d/server.cnf

[mysqld]
server-id=2
relay_log=mariadb-relay-bin
read_only=ON
log_bin=mariadb-bin
binlog_format=ROW
bind-address=0.0.0.0

systemctl restart mariadb



测试

主库已有 collect_db 数据

主库执行
mysqldump -uroot -p --single-transaction --master-data=2 --databases collect_db > /root/collect_db.sql

拷贝到：
scp /root/collect_db.sql root@192.168.1.11:/root/


从库导入
mysql -uroot -p < /root/collect_db.sql

查看 dump 文件中的主库 binlog 点位：
grep -m 1 "CHANGE MASTER" /root/collect_db.sql

-- CHANGE MASTER TO MASTER_LOG_FILE='mariadb-bin.000001', MASTER_LOG_POS=1234;

## 建议创建两个业务用户


1. 主库写入用户
CREATE USER 'collect_user'@'%' IDENTIFIED BY 'linux0224';

GRANT SELECT, INSERT, UPDATE, DELETE 
ON collect_db.* 
TO 'collect_user'@'%';

FLUSH PRIVILEGES;

2. 从库只读用户
CREATE USER 'readonly_user'@'%' IDENTIFIED BY 'linux0224';

GRANT SELECT 
ON collect_db.* 
TO 'readonly_user'@'%';

FLUSH PRIVILEGES;
